\begin{variablelist}
	
	% ==================== 罗马字母 ====================
	$d$          & 喷嘴孔径，mm \\
	$d_p$        & 示踪粒子直径，m \\
	$f$          & 激励频率，Hz \\
	$f_1, f_2$   & 混合物分数 \\
	$L$          & 流场特征长度，m \\
	$L_{RZ}$     & 回流区长度，m \\
	$L_{RZ}^{mean}$ & 回流区平均长度，m \\
	$R_1, R_2$   & 归一化红/蓝单色粒子图像强度值 \\
	$S_t$        & 斯特劳哈尔数 \\
	$St_k$       & 斯托克斯数 \\
	$U$          & 流场特征速度，m/s \\
	$U_j$        & 射流速度，m/s \\
	$U_m$        & 主流速度，m/s \\
	
	% ==================== 希腊字母 ====================
	$\lambda$    & 对流波长，m \\
	$\mu_f$      & 流体动力粘度，Pa·s \\
	$\rho_p$     & 示踪粒子密度，kg/m$^3$ \\
	$\tau_f$     & 流场特征时间，s \\
	$\tau_p$     & 粒子弛豫时间，s \\
	$\varphi(f)$ & 归一化熄火当量比 \\
	$\varnothing(f)$ & 受激励工况熄火当量比 \\
	$\varnothing_0$ & 无扰动工况熄火当量比 \\
	
	% ==================== 下标符号 ====================
	$f$          & 燃料 \\
	$j$          & 射流 \\
	$m$          & 主流 \\
	$p$          & 粒子 \\
	$RZ$         & 回流区 \\
	
	% ==================== 缩写词 ====================
	DMD          & 动态模态分解（Dynamic Mode Decomposition） \\
	FTF          & 火焰传递函数（Flame Transfer Function） \\
	LBO          & 贫燃吹熄（Lean Blowout） \\
	LIF          & 激光诱导荧光（Laser Induced Fluorescence） \\
	PIV          & 粒子图像测速（Particle Image Velocimetry） \\
	PLIF         & 平面激光诱导荧光（Planar Laser Induced Fluorescence） \\
	SLM          & 标准升每分钟（Standard Litre per Minute） \\
	
\end{variablelist}
